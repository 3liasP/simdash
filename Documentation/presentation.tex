\section{Introduction}

\begin{frame}{What is SimDash?}
\begin{tlist}
    \item Interactive driving simulator combining \alert{hardware} and \alert{software}
    \item Real-time feedback through physical components
    \item Unity-based 3D environment with realistic physics
    \item Arduino-controlled peripherals for immersive experience
\end{tlist}

\vspace{0.5cm}
\textbf{Core Concept:} Bridge the gap between virtual simulation and tangible interaction
\end{frame}

\begin{frame}{Why Driving Simulation?}
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Applications:}
\begin{tlist}
    \item Driver training and education
    \item Behavioral analysis in psychology
    \item Medical research (stress response)
    \item For fun!
\end{tlist}

\column{0.5\textwidth}
\textbf{Our Focus:}
\begin{tlist}
    \item Semi-realistic environmental simulation
    \item Tangible feedback systems
    \item Accessible prototyping approach
\end{tlist}
\end{columns}
\end{frame}

\section{Complete Interaction System}

\begin{frame}{System Architecture}
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Software (Unity):}
\begin{tlist}
    \item 3D driving environment
    \item Realistic physics engine
    \item Weather simulation (rain)
    \item Random event scenarios
    \item Serial communication interface
    \item Speed monitoring \& warnings
\end{tlist}

\column{0.5\textwidth}
\textbf{Hardware (Arduino):}
\begin{tlist}
    \item Arduino Uno microcontroller
    \item 3-digit 7-segment display (5641AS)
    \item 5V fan motor with PWM control
    \item Steering wheel input device
    \item USB serial connection
\end{tlist}
\end{columns}

\vspace{0.3cm}
\centering
\alert{Bidirectional data flow: Unity $\leftrightarrow$ Serial $\leftrightarrow$ Arduino}
\end{frame}

\begin{frame}{Hardware Components}
\begin{columns}[T]
\column{0.6\textwidth}
\textbf{7-Segment Display (5641AS):}
\begin{tlist}
    \item Displays speed (0-100)
    \item 3-digit common cathode configuration
    \item Segment pins: 9, 2, 3, 5, 6, 8, 7, 4
    \item Digit control: pins 10, 11, 12
    \item Uses SevSeg library for multiplexing
\end{tlist}

\textbf{Fan Motor:}
\begin{tlist}
    \item 5V DC motor
    \item PWM control via pin 13
    \item Speed mapped from game (0-100 → 0-255 PWM)
    \item Simulates wind flow during driving
\end{tlist}

\column{0.4\textwidth}
\begin{center}

\end{columns}
\end{frame}

\begin{columns}[T]

\section{Proposal vs. Final Product}

\begin{frame}{Original Vision}
\textbf{Initial Proposal Included:}
\begin{tlist}
    \item Heart rate sensor for stress monitoring
    \item ECG helmet for brain signal analysis
    \item Automatic speed control based on driver stress
    \item Behavioral analysis during driving
    \item Medical/psychological research applications
\end{tlist}

\vspace{0.3cm}
\begin{center}
\alert{Ambitious scope for rapid prototyping timeline}
\end{center}
\end{frame}

\begin{frame}{What Changed?}
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Removed Features:}
\begin{tlist}
    \item Heart rate sensor
    \begin{tlist}
        \item Delayed delivery
        \item Time constraints
    \end{tlist}
    \item ECG helmet integration
    \begin{tlist}
        \item Complex implementation
        \item Not available for student use
        \item Scope too large
    \end{tlist}
    \item Automatic stress-based control
\end{tlist}

\column{0.5\textwidth}
\textbf{Enhanced Instead:}
\begin{tlist}
    \item Polished driving environment
    \item Improved weather effects
    \item Better audio integration
    \item More stable hardware setup
    \item Refined physics simulation
    \item Random scenario system
\end{tlist}
\end{columns}

\vspace{0.5cm}
\centering
\textbf{Result:} More focused, achievable prototype that still demonstrates core concepts
\end{frame}

\section{Lessons Learned}

\begin{frame}{Team Insights: Technical}
\textbf{Key Technical Learnings:}
\begin{tlist}
    \item \alert{Unity Basics:} Several members learned Unity from scratch
    \item \alert{GitHub Proficiency:} Version control skills developed across team
    \item \alert{Serial Communication:} Understanding of hardware-software protocols
    \item \alert{Arduino Programming:} Microcontroller development experience
    \item \alert{Circuit Design:} Practical electronics knowledge gained
\end{tlist}

\vspace{0.5cm}
\textbf{Individual Reflections:}
\begin{tlist}
    \item Initial learning curve was steep for less experienced members
    \item Cross-learning between team members valuable
    \item Hands-on experience more effective than theoretical study
\end{tlist}
\end{frame}

\begin{frame}{Team Insights: Process}
\textbf{What Worked:}
\begin{tlist}
    \item Freedom to choose tasks based on interest/skill
    \item Early tool adoption (GitHub, Miro)
    \item Flexibility in adjusting scope
\end{tlist}

\vspace{0.3cm}
\textbf{Challenges Faced:}
\begin{tlist}
    \item \alert{Unclear task division} led to some confusion
    \item \alert{Limited prior experience} with technologies as a team
    \item \alert{Merge conflicts} from uncoordinated parallel work
\end{tlist}
\end{frame}

\begin{frame}{Rapid Prototyping Insights}
\textbf{Key Lessons:}

\begin{tlist}
    \item \alert{Hardware Risks:} Component availability is important from the start
    \item \alert{Integration Early:} Test hardware-software connection ASAP
    \item \alert{Parallel Work:} Not so simple with Unity Scenes to avoid conflicts
    \item \alert{Flexibility:} Be ready to pivot when obstacles arise
\end{tlist}

\vspace{0.5cm}
\begin{center}
\textbf{Despite challenges, we delivered a functional prototype\\
that demonstrates our core concept successfully!}
\end{center}
\end{frame}
